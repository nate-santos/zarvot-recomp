cmake_minimum_required(VERSION 3.20)
project(recomp_memory_gate LANGUAGES C CXX)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION FALSE)
add_executable(memory_gate_gen generate.cpp)
target_compile_features(memory_gate_gen PRIVATE cxx_std_20)
target_include_directories(memory_gate_gen PRIVATE "${EMITTER_INCLUDE}")
set(gen "${CMAKE_CURRENT_BINARY_DIR}/gen")
set(generated "${gen}/recomp_runtime.h")
foreach(tag baseline aligned reversed combined masked abi5)
  list(APPEND generated "${gen}/${tag}.c" "${gen}/${tag}_names.h")
endforeach()
add_custom_command(OUTPUT ${generated} COMMAND memory_gate_gen "${gen}"
                   DEPENDS memory_gate_gen VERBATIM)
add_custom_target(memory_gate_generated DEPENDS ${generated})
foreach(tag baseline aligned reversed combined masked abi5)
  add_library(memory_${tag} STATIC "${gen}/${tag}.c")
  add_dependencies(memory_${tag} memory_gate_generated)
  target_compile_features(memory_${tag} PRIVATE c_std_11)
  target_include_directories(memory_${tag} PRIVATE "${gen}")
  target_compile_definitions(memory_${tag} PRIVATE SUYU_HOSTED_RECOMP=1 RECOMP_STATIC_HOST=1)
  list(APPEND compiled_targets memory_${tag})
endforeach()
add_executable(memory_gate runner.cpp)
add_dependencies(memory_gate memory_gate_generated)
target_compile_features(memory_gate PRIVATE cxx_std_20)
target_include_directories(memory_gate PRIVATE "${gen}")
target_compile_definitions(memory_gate PRIVATE SUYU_HOSTED_RECOMP=1 RECOMP_STATIC_HOST=1)
target_link_libraries(memory_gate PRIVATE memory_baseline memory_aligned memory_reversed memory_combined memory_masked)
foreach(tag baseline aligned reversed combined masked)
  add_executable(fmdiff_${tag} "${FMDIFF_SOURCE}")
  add_dependencies(fmdiff_${tag} memory_gate_generated)
  target_compile_features(fmdiff_${tag} PRIVATE c_std_11)
  target_include_directories(fmdiff_${tag} PRIVATE "${gen}")
  target_compile_definitions(fmdiff_${tag} PRIVATE SUYU_HOSTED_RECOMP=1 RECOMP_STATIC_HOST=1)
  target_link_libraries(fmdiff_${tag} PRIVATE memory_${tag} memory_abi5)
  if(MSVC)
    target_compile_options(fmdiff_${tag} PRIVATE "/FI${gen}/${tag}_names.h")
  else()
    target_compile_options(fmdiff_${tag} PRIVATE -include "${gen}/${tag}_names.h")
    target_link_libraries(fmdiff_${tag} PRIVATE m)
  endif()
  list(APPEND compiled_targets fmdiff_${tag})
endforeach()
list(APPEND compiled_targets memory_gate memory_gate_gen)
foreach(target IN LISTS compiled_targets)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /Ob2 /bigobj /fp:precise)
  else()
    target_compile_options(${target} PRIVATE -O2)
  endif()
endforeach()
if(NOT MSVC)
  target_link_libraries(memory_gate PRIVATE m)
endif()
